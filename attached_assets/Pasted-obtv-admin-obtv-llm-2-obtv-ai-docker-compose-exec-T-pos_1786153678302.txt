obtv-admin@obtv-llm-2:~/obtv-ai$ docker compose exec -T postgres psql -U obtv -d obtv <<'SQL'
BEGIN;
CREATE TEMP TABLE doomed AS
  SELECT id FROM media_assets
  WHERE original_path LIKE '/curator/%' AND status IN ('pending','processing');
DO $$
DECLARE r RECORD;
BEGIN
  FOR pass IN 1..2 LOOP
    FOR r IN
      SELECT c.conrelid::regclass AS tbl, a.attname AS col
      FROM pg_constraint c
      JOIN pg_attribute a ON a.attrelid = c.conrelid AND a.attnum = ANY (c.conkey)
      WHERE c.confrelid = 'media_assets'::regclass AND c.contype = 'f'
    LOOP
      BEGIN
        EXECUTE format('DELETE FROM %s WHERE %I IN (SELECT id FROM doomed)', r.tbl, r.col);
      EXCEPTION WHEN foreign_key_violation THEN
        -- child of a child still holds rows; second pass will get it
        NULL;
      END;
    END LOOP;
  END LOOP;
END $$;
DELETE FROM media_assets WHERE id IN (SELECT id FROM doomed);
COMMIT;
